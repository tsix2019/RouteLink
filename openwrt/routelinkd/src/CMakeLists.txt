cmake_minimum_required(VERSION 3.13)
project(routelinkd C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-Wall -Wextra -Werror -Wno-unused-parameter)

set(RL_VERSION "0.0.0-dev" CACHE STRING "Version reported over ubus")
set(RL_BUILD_EPOCH "0" CACHE STRING "Build time (SOURCE_DATE_EPOCH): the clock is not trusted before it")
add_compile_definitions(RL_VERSION="${RL_VERSION}" RL_BUILD_EPOCH=${RL_BUILD_EPOCH})

option(RL_TESTS "Build host unit tests (core only)" OFF)
option(RL_SANITIZE "AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if(RL_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB CORE_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/core/*.c)
add_library(rlcore STATIC ${CORE_SOURCES})
target_include_directories(rlcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(rlcore json-c)

if(RL_TESTS)
  enable_testing()
  file(GLOB TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.c)
  foreach(src ${TEST_SOURCES})
    get_filename_component(name ${src} NAME_WE)
    add_executable(${name} ${src})
    target_link_libraries(${name} rlcore)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
else()
  file(GLOB SYS_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/sys/*.c)
  add_executable(routelinkd main.c ${SYS_SOURCES})
  target_link_libraries(routelinkd rlcore ubox ubus uci mnl blobmsg_json)
  install(TARGETS routelinkd RUNTIME DESTINATION sbin)
endif()
